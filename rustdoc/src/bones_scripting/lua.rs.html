61/// Resource containing the lua plugins that have been installed in this session.
62///
63/// Excluded from [`World`] serialization via [`SkipSerialize`]: this is local runtime state
64/// (loaded-script handles) that every machine sets up itself, not simulation state.
65#[cfg(not(target_os = "emscripten"))]
66#[derive(HasSchema, Deref, DerefMut, Default, Clone)]
67#[type_data(SkipSerialize)]
68pub struct LuaPlugins(pub Arc<Vec<Handle<LuaPlugin>>>);
69
70#[cfg(not(target_os = "emscripten"))]
71impl SessionPlugin for LuaPluginLoaderSessionPlugin {
72    fn install(self, session: &mut SessionBuilder) {
73        session.insert_resource(LuaPlugins(self.0));
74
75        for lua_stage in [
76            CoreStage::First,
77            CoreStage::PreUpdate,
78            CoreStage::Update,
79            CoreStage::PostUpdate,
80            CoreStage::Last,
81        ] {
82            session.stages.add_system_to_stage(
83                lua_stage,
84                move |engine: Res<LuaEngine>,
85                      asset_server: Res<AssetServer>,
86                      lua_plugins: Res<LuaPlugins>,
87                      world: &World| {
88                    engine.exec(|lua| {
89                        Frozen::<Freeze![&'freeze World]>::in_scope(world, |world| {
90                            lua.enter(|ctx| {
91                                let env = ctx.singletons().get(ctx, bindings::env);
92                                let worldref = WorldRef(world);
93                                worldref.add_to_env(ctx, env);
94                            });
95
96                            for plugin_handle in lua_plugins.iter() {
97                                let Some(plugin) = asset_server.try_get(*plugin_handle) else {
98                                    return;
99                                };
100                                let plugin = plugin.unwrap();
101
102                                // Load the plugin if necessary
103                                if !plugin.has_loaded() {
104                                    if let Err(e) = plugin.load(engine.executor.clone(), lua) {
105                                        tracing::error!("Error loading lua plugin: {e}");
106                                    }
107                                }
108
109                                let mut systems = plugin.systems.borrow_mut();
110                                let systems = systems.as_loaded_mut();
111
112                                for (has_run, closure) in &mut systems.startup {
113                                    if !*has_run {
114                                        let executor = lua.enter(|ctx| {
115                                            let closure = ctx.registry().fetch(closure);
116                                            let ex = Executor::start(ctx, closure.into(), ());
117                                            ctx.registry().stash(&ctx, ex)
118                                        });
119                                        if let Err(e) = lua.execute::<()>(&executor) {
120                                            tracing::error!("Error running lua plugin system: {e}");
121                                        }
122
123                                        *has_run = true;
124                                    }
125                                }
126
127                                for (stage, closure) in &systems.core_stages {
128                                    if stage == &lua_stage {
129                                        let executor = lua.enter(|ctx| {
130                                            let closure = ctx.registry().fetch(closure);
131                                            let ex = Executor::start(ctx, closure.into(), ());
132                                            ctx.registry().stash(&ctx, ex)
133                                        });
134                                        if let Err(e) = lua.execute::<()>(&executor) {
135                                            tracing::error!("Error running lua plugin system: {e}");
136                                        }
137                                    }
138                                }
139                            }
140                        })
141                    });
142                },
143            );
144        }
145    }
146}

149///
150// This type can be converted into lua userdata for accessing the world from lua.

153impl Default for WorldRef {

158impl<'gc> FromValue<'gc> for &'gc WorldRef {
159    fn from_value(_ctx: Context<'gc>, value: Value<'gc>) -> Result<Self, piccolo::TypeError> {
160        value.as_static_user_data::<WorldRef>()
161    }

164impl WorldRef {
165    /// Convert this [`WorldRef`] into a Lua userdata.
166    pub fn into_userdata(self, ctx: Context<'_>) -> UserData<'_> {
167        let data = UserData::new_static(&ctx, self);
168        data.set_metatable(
169            &ctx,
170            Some(ctx.singletons().get(ctx, bindings::world::metatable)),
171        );
172        data
173    }
174
175    /// Add this world to the lua environment (globals + env userdata for world/components/
176    /// resources/assets) — public so external stage runners can replicate the session plugin.
177    pub fn add_to_env<'gc>(&self, ctx: Context<'gc>, env: Table<'gc>) {
178        ctx.globals()
179            .set(ctx, "world", self.clone().into_userdata(ctx))
180            .unwrap();
181        for (name, metatable) in [
182            ("world", bindings::world::metatable as fn(Context) -> Table),
183            ("components", bindings::components::metatable),
184            ("resources", bindings::resources::metatable),
185            #[cfg(not(target_os = "emscripten"))]
186            ("assets", bindings::assets::metatable),
187        ] {
188            let data = UserData::new_static(&ctx, self.clone());
189            data.set_metatable(&ctx, Some(ctx.singletons().get(ctx, metatable)));
190            env.set(ctx, name, data).unwrap();
191        }
192    }

195/// Resource used to access the lua scripting engine.
196///
197/// Excluded from [`World`] serialization via [`SkipSerialize`]: the engine (Lua VM, executor,
198/// compiled-script cache) is opaque, local runtime state that each machine rebuilds itself from the
199/// same scripts. It is not simulation state and must never be sent over the wire. All authoritative
200/// gameplay state must live in Rust-typed components, never in Lua-side globals.
201#[derive(HasSchema, Clone)]
202#[schema(no_default)]
203#[type_data(SkipSerialize)]
204pub struct LuaEngine {
205    /// The thread-local task executor that is used to spawn any tasks that need access to the
206    /// lua engine which can only be accessed on it's own thread.
207    executor: ScriptExecutor,
208    /// The lua engine state container.
209    state: Arc<SendWrapper<EngineState>>,
210}
211
212/// Internal state for [`LuaEngine`].
213struct EngineState {
214    /// The Lua engine.
215    lua: Mutex<Lua>,
216    /// Persisted lua data we need stored in Rust, such as the environment table, world
217    /// metatable, etc.
218    data: LuaSingletons,
219    /// Cache of the content IDs of loaded scripts, and their compiled lua closures.
220    #[cfg(not(target_os = "emscripten"))]
221    compiled_scripts: Mutex<HashMap<Cid, StashedClosure>>,
222}

224// TODO: Don't Use Function Pointers to Index Lua Singletons.
225// Unfortunately, function pointers with different signatures may be unified by
226// LLVM when activating, for intance, LTO, and that can cause unexpected behavior
227// when using them as indexes into a HashMap for singletons.
228/// Struct for accessing and initializing lua singletons.
229///
230/// This is stored in a lua global and accessed conveniently through our [`CtxExt`] trait
231/// so that we can easily initialize lua tables and callbacks throughout our lua bindings.
232pub struct LuaSingletons {
233    singletons: Rc<AtomicCell<HashMap<usize, Box<dyn Any>>>>,
234}
235impl Default for LuaSingletons {
236    fn default() -> Self {
237        Self {
238            singletons: Rc::new(AtomicCell::new(HashMap::default())),
239        }
240    }
241}
242impl LuaSingletons {
243    /// Fetch a lua singleton, initializing it if it has not yet been created.
244    ///
245    /// The singleton is defined by a function pointer that returns a stashable value.
246    /// Public so external stage runners (embedding the lua machinery without
247    /// `bones_lib::Session`) can fetch the shared environment/metatables.
248    pub fn get<
249        'gc,
250        S: Fetchable<'gc, Fetched = T> + 'static,
251        T: Stashable<'gc, Stashed = S> + Clone + Copy + 'gc,
252    >(
253        &self,
254        ctx: Context<'gc>,
255        singleton: fn(Context<'gc>) -> T,
256    ) -> T {
257        let map = self.singletons.borrow_mut();
258        let id = singleton as usize;
259        if let Some(entry) = map.get(&id) {
260            let stashed = entry.downcast_ref::<S>().expect(
261                "Encountered two functions with different return types and \
262                the same function pointer.",
263            );
264            ctx.registry().fetch(stashed)
265        } else {
266            drop(map); // Make sure we don't deadlock
267            let v = singleton(ctx);
268            let stashed = ctx.registry().stash(&ctx, v);
269            self.singletons.borrow_mut().insert(id, Box::new(stashed));
270            v
271        }
272    }
273}

275impl Default for EngineState {
276    fn default() -> Self {
277        // Initialize an empty lua engine and our lua data.
278        let mut lua = Lua::core();
279        lua.try_enter(|ctx| {
280            // Insert our lua singletons.
281            ctx.globals().set(
282                ctx,
283                "luasingletons",
284                UserData::new_static(&ctx, LuaSingletons::default()),
285            )?;
286            Ok(())
287        })
288        .unwrap();
289        Self {
290            lua: Mutex::new(lua),
291            data: default(),
292            #[cfg(not(target_os = "emscripten"))]
293            compiled_scripts: default(),
294        }
295    }
296}

298impl Default for LuaEngine {
299    /// Initialize the Lua engine.
300    fn default() -> Self {
301        // Make sure the compute task pool is initialized
302        #[cfg(not(target_arch = "wasm32"))]
303        ComputeTaskPool::init(TaskPool::new);
304
305        #[cfg(not(target_arch = "wasm32"))]
306        let executor = {
307            let (send, recv) = async_channel::bounded(1);
308
309            // Spawn the executor task that will be used for the lua engine.
310            let pool = ComputeTaskPool::get();
311            pool.spawn_local(async move {
312                let executor = Arc::new(ThreadExecutor::new());
313                send.try_send(executor.clone()).unwrap();
314
315                let ticker = (*executor).ticker().unwrap();
316                loop {
317                    ticker.tick().await;
318                }
319            })
320            .detach();
321            pool.with_local_executor(|local| while local.try_tick() {});
322
323            recv.try_recv().unwrap()
324        };
325
326        #[cfg(target_arch = "wasm32")]
327        let executor = ScriptExecutor;
328
329        LuaEngine {
330            executor,
331            state: Arc::new(SendWrapper::new(default())),
332        }
333    }
334}

336impl LuaEngine {
337    /// The engine's dedicated executor — external stage runners (games embedding the lua
338    /// machinery without `bones_lib::Session`) need it to call [`LuaPlugin::load`].
339    pub fn executor(&self) -> &ScriptExecutor {
340        &self.executor
341    }
342
343    /// Access the lua engine to run code on it.
344    #[cfg(not(target_arch = "wasm32"))]
345    pub fn exec<'a, F: FnOnce(&mut Lua) + Send + 'a>(&self, f: F) {
346        let pool = ComputeTaskPool::get();
347
348        // Create a new scope spawned on the lua engine thread.
349        pool.scope_with_executor(false, Some(&self.executor), |scope| {
350            scope.spawn_on_external(async {
351                f(&mut self.state.lua.lock());
352            });
353        });
354    }
355
356    /// Access the lua engine to run code on it.
357    ///
358    /// wasm runs the closure directly: the build is single-threaded, the engine lives on the
359    /// (only) main thread, and bevy_tasks is not linked on wasm at all (its wasm path is
360    /// wasm-bindgen-based, which emscripten cannot load — see Cargo.toml). Behaviorally this
361    /// matches what bevy_tasks' single-threaded pool would have done anyway: its
362    /// `scope_with_executor` ignores the thread executor and drives spawned tasks to completion
363    /// inline on a `LocalExecutor`.
364    #[cfg(target_arch = "wasm32")]
365    pub fn exec<'a, F: FnOnce(&mut Lua) + Send + 'a>(&self, f: F) {
366        f(&mut self.state.lua.lock());
367    }
368
369    /// Run a lua script as a system on the given world.
370    #[cfg(not(target_os = "emscripten"))]
371    pub fn run_script_system(&self, world: &World, script: Handle<LuaScript>) {
372        self.exec(|lua| {
373            Frozen::<Freeze![&'freeze World]>::in_scope(world, |world| {
374                // Wrap world reference so that it can be converted to lua userdata.
375                let worldref = WorldRef(world);
376
377                let executor = lua.try_enter(|ctx| {
378                    // Fetch the env table
379                    let env = self.state.data.get(ctx, bindings::env);
380
381                    // Compile the script
382                    let closure = worldref.with(|world| {
383                        let asset_server = world.resource::<AssetServer>();
384                        let cid = *asset_server
385                            .store
386                            .asset_ids
387                            .get(&script.untyped())
388                            .ok_or_else(|| {
389                                tracing::warn!("Script asset not loaded.");
390                                PrototypeError::Parser(ParseError {
391                                    kind: piccolo::compiler::ParseErrorKind::EndOfStream {
392                                        expected: None,
393                                    },
394                                    line_number: LineNumber(0),
395                                })
396                            })?;
397
398                        let mut compiled_scripts = self.state.compiled_scripts.lock();
399                        let closure = compiled_scripts.get(&cid);
400
401                        Ok::<_, PrototypeError>(match closure {
402                            Some(closure) => ctx.registry().fetch(closure),
403                            None => {
404                                let asset = asset_server.store.assets.get(&cid).unwrap();
405                                let source = &asset.data.cast_ref::<LuaScript>().source;
406                                // TODO: Provide a meaningfull name to loaded scripts.
407                                let closure =
408                                    Closure::load_with_env(ctx, None, source.as_bytes(), env)?;
409                                compiled_scripts.insert(cid, ctx.registry().stash(&ctx, closure));
410
411                                closure
412                            }
413                        })
414                    })?;
415
416                    // Insert the world ref into the global scope
417                    worldref.add_to_env(ctx, env);
418
419                    let ex = Executor::start(ctx, closure.into(), ());
420                    let ex = ctx.registry().stash(&ctx, ex);
421                    Ok(ex)
422                });
423
424                if let Err(e) = executor.and_then(|ex| lua.execute::<()>(&ex)) {
425                    tracing::error!("{e}");
426                }
427            });
428        });
429    }
