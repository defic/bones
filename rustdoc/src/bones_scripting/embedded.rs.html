1//! Embedded / runtime-installed script plugins WITHOUT the asset server — for games that drive
2//! bare [`SystemStages`] (e.g. under an external netcode loop) and distribute script sources
3//! themselves (compiled-in defaults + content-addressed downloads).
4//!
5//! Flow (mirrors an ADR-0005-style "slot"): the host calls [`set_active_source`] with verified
6//! bytes BEFORE constructing a session (server boot / client prefetch), then builds the session's
7//! [`ScriptPlugins`] resource with [`plugins_for_session`] (which falls back to the game's
8//! compiled-in default source) and installs the per-stage runners with [`install_lua_runners`].
9//!
10//! Determinism rules for scripts running inside a predicted, rolled-back tick:
11//! - every peer must run byte-identical sources (content-address them!);
12//! - ALL persistent state lives in reflected ECS resources/components — the Lua VM is never
13//!   snapshotted, so Lua globals are per-tick scratch at best.
14

23/// The loaded Lua plugins for a session, in registration order. Shared `Arc` so rollback clones
24/// reuse the same compiled plugins (their `systems` cell is shared — startup closures run once
25/// per session, not once per clone). `labels` parallels `plugins` (same order) — human names
26/// for diagnostics like the per-script [`StageTimings`] rows. `SkipSerialize`: runtime state
27/// every peer builds itself.
28#[derive(HasSchema, Clone, Default)]
29#[schema(opaque)]
30#[type_data(SkipSerialize)]
31pub struct ScriptPlugins {
32    pub plugins: Arc<Vec<Arc<LuaPlugin>>>,
33    pub labels: Arc<Vec<String>>,
34}
35
36struct SourceSlot {
37    /// The ORDERED `(source, label)`s pending for the next session. `None` = "use the game's
38    /// default"; `Some(vec)` = exactly these (each file becomes its own plugin, in order —
39    /// scripts self-schedule onto stages, files are never flattened together).
40    pending: Option<Vec<(String, String)>>,
41    active: Option<Arc<Vec<(String, String)>>>,
42    consumed: bool,
43}
44
45fn slot() -> &'static Mutex<SourceSlot> {
46    static SLOT: OnceLock<Mutex<SourceSlot>> = OnceLock::new();
47    SLOT.get_or_init(|| Mutex::new(SourceSlot { pending: None, active: None, consumed: false }))
48}
49
50/// Append one runtime script source for the NEXT session (call once per file, in run order —
51/// the client prefetch and the server boot both install files individually so each stays its
52/// own cacheable, content-addressed asset). Returns `false` if a session already baked the
53/// pending set (the install came too late — surface it, peers may diverge). Non-UTF-8 rejected.
54pub fn push_active_source(bytes: Vec<u8>) -> bool {
55    push_active_source_labeled(bytes, "")
56}
57
58/// [`push_active_source`] with a human label for diagnostics (per-script timing rows, logs).
59/// An empty label falls back to `file-<index>`.
60pub fn push_active_source_labeled(bytes: Vec<u8>, label: &str) -> bool {
61    let Ok(source) = String::from_utf8(bytes) else {
62        tracing::error!("push_active_source: script asset is not UTF-8 — ignoring");
63        return false;
64    };
65    let mut slot = slot().lock().unwrap();
66    let in_time = !slot.consumed;
67    if slot.consumed {
68        // A stale pending set must not leak into the next install sequence.
69        slot.pending = None;
70        slot.consumed = false;
71    }
72    let pending = slot.pending.get_or_insert_with(Vec::new);
73    let label = if label.is_empty() { format!("file-{}", pending.len()) } else { label.to_string() };
74    pending.push((source, label));
75    in_time
76}

78/// Replace the pending set with exactly one source (single-script convenience).
79pub fn set_active_source(bytes: Vec<u8>) -> bool {
80    {
81        let mut slot = slot().lock().unwrap();
82        slot.pending = None;
83        slot.consumed = false;
84    }
85    push_active_source(bytes)
86}
87
88/// Back to the game's compiled-in default source for the NEXT session.
89pub fn reset_active_source() {
90    let mut slot = slot().lock().unwrap();
91    slot.pending = None;
92    slot.consumed = false;
93}

95/// Commit the pending source set for this session (`default_source` when none installed) and
96/// build the session's [`ScriptPlugins`] — one plugin per file, in install order. Also installs
97/// (once per process) the lua type data the stock session plugin would have registered —
98/// without it `Entities` has no `iter_with` metatable and every script query fails.
99pub fn plugins_for_session(default_source: &str) -> ScriptPlugins {
100    static LUA_TYPEDATA: Once = Once::new();
101    LUA_TYPEDATA.call_once(bindings::register_lua_typedata);
102
103    let sources = {
104        let mut slot = slot().lock().unwrap();
105        if slot.consumed {
106            if let Some(active) = &slot.active {
107                active.clone()
108            } else {
109                let s = Arc::new(vec![(default_source.to_string(), "default".to_string())]);
110                slot.active = Some(s.clone());
111                s
112            }
113        } else {
114            let s = Arc::new(match slot.pending.take() {
115                Some(sources) if !sources.is_empty() => sources,
116                _ => vec![(default_source.to_string(), "default".to_string())],
117            });
118            slot.active = Some(s.clone());
119            slot.consumed = true;
120            s
121        }
122    };
123    ScriptPlugins {
124        plugins: Arc::new(
125            sources
126                .iter()
127                .map(|(source, _)| {
128                    Arc::new(LuaPlugin {
129                        source: source.clone(),
130                        systems: Arc::new(AtomicCell::new(Default::default())),
131                    })
132                })
133                .collect(),
134        ),
135        labels: Arc::new(sources.iter().map(|(_, label)| label.clone()).collect()),
136    }
137}

139/// Register a Lua runner system at the END of every core stage of `builder` (after that stage's
140/// host systems, by insertion order). Fork-variant of the stock `LuaPluginLoaderSessionPlugin`
141/// body, reading plugins from the [`ScriptPlugins`] resource instead of asset handles.
142pub fn install_lua_runners(builder: &mut SystemStagesBuilder) {
143    for lua_stage in [
144        CoreStage::First,
145        CoreStage::PreUpdate,
146        CoreStage::Update,
147        CoreStage::PostUpdate,
148        CoreStage::Last,
149    ] {
150        // Named explicitly: the closure would otherwise report THIS function's name in the
151        // per-system timing table, reading as if "install" ran every frame.
152        let mut runner = IntoSystem::system(
153            move |engine: Res<LuaEngine>, scripts: Res<ScriptPlugins>, world: &World| {
154                // Dormant fast path: the `engine.exec` hop costs microseconds×allocs, so skip it
155                // unless some plugin actually needs this stage (still loading, pending startup,
156                // or has a closure registered here).
157                let stage_has_work = scripts.plugins.iter().any(|p| {
158                    let systems = p.systems.borrow();
159                    match &*systems {
160                        LuaPluginSystemsState::NotLoaded => true,
161                        LuaPluginSystemsState::Loaded { .. } => {
162                            let s = systems.as_loaded();
163                            s.startup.iter().any(|(has_run, _)| !has_run)
164                                || s.core_stages.iter().any(|(stage, _)| *stage == lua_stage)
165                        }
166                        LuaPluginSystemsState::Unloaded => false,
167                    }
168                });
169                if !stage_has_work {
170                    return;
171                }
172                // Per-script timing (see bones_ecs `StageTimings`): only when an enabled
173                // resource is present, and recorded AFTER the VM scope ends (the resource
174                // must not be borrowed while scripts run).
175                let timing = world
176                    .resources
177                    .get::<StageTimings>()
178                    .map(|t| t.enabled)
179                    .unwrap_or(false);
180                let mut plugin_ns: Vec<u64> = if timing {
181                    vec![0; scripts.plugins.len()]
182                } else {
183                    Vec::new()
184                };
185                engine.exec(|lua| {
186                    Frozen::<Freeze![&'freeze World]>::in_scope(world, |world| {
187                        lua.enter(|ctx| {
188                            let env = ctx.singletons().get(ctx, bindings::env);
189                            let worldref = WorldRef(world);
190                            worldref.add_to_env(ctx, env);
191                        });
192
193                        for (plugin_idx, plugin) in scripts.plugins.iter().enumerate() {
194                            let t0 = timing.then(std::time::Instant::now);
195                            if !plugin.has_loaded() {
196                                if let Err(e) = plugin.load(engine.executor().clone(), lua) {
197                                    eprintln!("lua plugin load error: {e}");
198                                    tracing::error!("Error loading lua plugin: {e}");
199                                }
200                            }
201                            if matches!(&*plugin.systems.borrow(), LuaPluginSystemsState::NotLoaded)
202                            {
203                                continue;
204                            }
205
206                            let mut systems = plugin.systems.borrow_mut();
207                            let systems = systems.as_loaded_mut();
208
209                            for (has_run, closure) in &mut systems.startup {
210                                if !*has_run {
211                                    let executor = lua.enter(|ctx| {
212                                        let closure = ctx.registry().fetch(closure);
213                                        let ex = crate::lua::piccolo::Executor::start(
214                                            ctx,
215                                            closure.into(),
216                                            (),
217                                        );
218                                        ctx.registry().stash(&ctx, ex)
219                                    });
220                                    if let Err(e) = lua.execute::<()>(&executor) {
221                                        eprintln!("lua startup system error: {e}");
222                                        tracing::error!("Error running lua startup system: {e}");
223                                    }
224                                    *has_run = true;
225                                }
226                            }
227
228                            for (stage, closure) in &systems.core_stages {
229                                if stage == &lua_stage {
230                                    let executor = lua.enter(|ctx| {
231                                        let closure = ctx.registry().fetch(closure);
232                                        let ex = crate::lua::piccolo::Executor::start(
233                                            ctx,
234                                            closure.into(),
235                                            (),
236                                        );
237                                        ctx.registry().stash(&ctx, ex)
238                                    });
239                                    if let Err(e) = lua.execute::<()>(&executor) {
240                                        eprintln!("lua system error: {e}");
241                                        tracing::error!("Error running lua system: {e}");
242                                    }
243                                }
244                            }
245                            if let Some(t0) = t0 {
246                                plugin_ns[plugin_idx] += t0.elapsed().as_nanos() as u64;
247                            }
248                        }
249                    })
250                });
251                if timing {
252                    if let Some(mut t) = world.resources.get_mut::<StageTimings>() {
253                        let stage_name = lua_stage.name();
254                        for (i, ns) in plugin_ns.iter().enumerate() {
255                            if *ns > 0 {
256                                let label = scripts
257                                    .labels
258                                    .get(i)
259                                    .map(|l| l.as_str())
260                                    .unwrap_or("?");
261                                t.record(&stage_name, &format!("lua:{label}"), *ns);
262                            }
263                        }
264                    }
265                }
266            },
267        );
268        runner.name = "lua_runner";
269        builder.add_system_to_stage(lua_stage, runner);
270    }
271}

273/// One-shot console chunks queued by the host — e.g. an admin "eval" arriving as a REPLICATED
274/// game input, pushed here by the host's resolve system and drained the same run by the
275/// runner from [`install_eval_runner`]. `SkipSerialize` scratch: determinism comes from the
276/// chunks being replicated inputs (every peer queues the same sources at the same tick), not
277/// from this resource riding the snapshot.
278#[derive(HasSchema, Clone, Default)]
279#[schema(opaque)]
280#[type_data(SkipSerialize)]
281pub struct EvalQueue(pub Vec<String>);
282
283/// One formatted result line per executed [`EvalQueue`] chunk, in execution order — every
284/// peer computes them locally (same chunks, same state), so the server can answer the admin
285/// request AND an in-client console can render them without any extra wire traffic. Capped
286/// FIFO (a peer with no console just cycles the buffer). `SkipSerialize`: presentation.
287#[derive(HasSchema, Clone, Default)]
288#[schema(opaque)]
289#[type_data(SkipSerialize)]
290pub struct EvalResults(pub Vec<String>);
291
292/// Most results a peer keeps before dropping the oldest.
293const EVAL_RESULTS_CAP: usize = 32;
294
295/// Game-defined console prelude: Lua PREPENDED to every [`eval_chunk`] source at COMPILE time
296/// (never on the wire — a replicated console input stays just the user's line). The intended
297/// use is `local` aliases that make console lines terse:
298/// `local tune = resources:get(s("Tune"))` → users type `tune.race.laps = 5`.
299/// Insert it as a resource at session build; identical on every peer by construction (it's
300/// compiled-in game code). `SkipSerialize`: static config, not state.
301#[derive(HasSchema, Clone, Default)]
302#[schema(opaque)]
303#[type_data(SkipSerialize)]
304pub struct EvalPrelude(pub String);
305
306/// Format a Lua value for the console: primitives natively; tables/functions/userdata as a
307/// type tag (calling `__tostring` metamethods would need another executor pump — reflected
308/// field reads land here as primitives anyway, which is the case that matters).
309fn fmt_lua_value(v: &crate::lua::piccolo::Value) -> String {
310    use crate::lua::piccolo::Value;
311    match v {
312        Value::Nil => "nil".to_string(),
313        Value::Boolean(b) => b.to_string(),
314        Value::Integer(i) => i.to_string(),
315        Value::Number(n) => n.to_string(),
316        Value::String(s) => String::from_utf8_lossy(s.as_bytes()).into_owned(),
317        Value::Table(_) => "<table>".to_string(),
318        Value::Function(_) => "<function>".to_string(),
319        Value::Thread(_) => "<thread>".to_string(),
320        Value::UserData(_) => "<userdata>".to_string(),
321    }
322}

324/// Compile + execute ONE console chunk against `world` (which must carry a [`LuaEngine`]
325/// resource) and return its formatted result line. An expression chunk is auto-wrapped as
326/// `return <src>` (fallback: plain statement chunk, result "ok"), so `…race.laps` answers
327/// with the value. Errors come back as `error: …` — never a panic.
328///
329/// Two callers, two determinism postures:
330/// - the replicated eval runner (below) — every peer runs the same chunk on the same state;
331/// - a host's read-only QUERY path — run it on a CLONE of the state, so a chunk that
332///   accidentally writes mutates the discarded clone, not the live sim.
333pub fn eval_chunk(world: &World, src: &str) -> String {
334    let Some(engine) = world.resources.get::<LuaEngine>() else {
335        return "error: no LuaEngine in world".to_string();
336    };
337    let prelude = world
338        .resources
339        .get::<EvalPrelude>()
340        .map(|p| p.0.clone())
341        .unwrap_or_default();
342    let mut result = String::new();
343    engine.exec(|lua| {
344        Frozen::<Freeze![&'freeze World]>::in_scope(world, |world| {
345            lua.enter(|ctx| {
346                let env = ctx.singletons().get(ctx, bindings::env);
347                let worldref = WorldRef(world);
348                worldref.add_to_env(ctx, env);
349            });
350            // Game prelude (alias locals), then: expression first (captures a return value),
351            // statement as fallback.
352            let with_return = format!("{prelude}\nreturn {src}");
353            let plain = format!("{prelude}\n{src}");
354            let executor = lua.try_enter(|ctx| {
355                let env = ctx.singletons().get(ctx, bindings::env);
356                let closure = crate::lua::piccolo::Closure::load_with_env(
357                    ctx,
358                    None,
359                    with_return.as_bytes(),
360                    env,
361                )
362                .or_else(|_| {
363                    crate::lua::piccolo::Closure::load_with_env(ctx, None, plain.as_bytes(), env)
364                })?;
365                let ex = crate::lua::piccolo::Executor::start(ctx, closure.into(), ());
366                Ok(ctx.registry().stash(&ctx, ex))
367            });
368            let line = executor.and_then(|ex| {
369                lua.finish(&ex);
370                lua.try_enter(|ctx| {
371                    let ex = ctx.registry().fetch(&ex);
372                    let vals = ex.take_result::<crate::lua::piccolo::Variadic<
373                        Vec<crate::lua::piccolo::Value>,
374                    >>(ctx)??;
375                    Ok(if vals.is_empty() {
376                        "ok".to_string()
377                    } else {
378                        vals.iter().map(fmt_lua_value).collect::<Vec<_>>().join(", ")
379                    })
380                })
381            });
382            result = match line {
383                Ok(s) => s,
384                Err(e) => {
385                    tracing::error!("lua eval error: {e}");
386                    format!("error: {e}")
387                }
388            };
389        })
390    });
391    result
392}

394/// Register the console-eval runner: drains [`EvalQueue`], executing each chunk via
395/// [`eval_chunk`] and pushing one result line per chunk into [`EvalResults`]. Install it
396/// AFTER the host's resolve systems on the same stage, so a chunk queued while applying a
397/// batch runs inside that same authoritative step on every peer. A bad chunk becomes an
398/// `error: …` line — never kills the sim, and errors identically on every peer.
399pub fn install_eval_runner(builder: &mut SystemStagesBuilder) {
400    let mut runner = IntoSystem::system(
401        move |world: &World| {
402            let chunks: Vec<String> = {
403                let Some(mut q) = world.resources.get_mut::<EvalQueue>() else {
404                    return;
405                };
406                if q.0.is_empty() {
407                    return;
408                }
409                std::mem::take(&mut q.0)
410            };
411            let results: Vec<String> =
412                chunks.iter().map(|src| eval_chunk(world, src)).collect();
413            if let Some(mut r) = world.resources.get_mut::<EvalResults>() {
414                r.0.extend(results);
415                let overflow = r.0.len().saturating_sub(EVAL_RESULTS_CAP);
416                if overflow > 0 {
417                    r.0.drain(..overflow);
418                }
419            }
420        },
421    );
422    runner.name = "lua_eval_console";
423    builder.add_system_to_stage(CoreStage::Update, runner);
424}

426/// Register the CONFIRMED-phase Lua runner: appended after the host's resolve system(s), it
427/// executes every plugin's `add_resolve_system` closures once per authoritative batch, in
428/// plugin order — post-application, so hooks see the batch's effects (joins applied, kills
429/// logged) atomically. Effects are confirmed-tick-late and never mispredicted.
430///
431/// v1 contract: resolve hooks read/write reflected state directly; outbound channels the game
432/// drains after its SIMULATE stages (e.g. a damage queue) are NOT re-drained here — pushing
433/// into them from a resolve hook does nothing until the game adds a resolve-side pipeline.
434pub fn install_resolve_lua_runner(builder: &mut SystemStagesBuilder) {
435    let mut runner = IntoSystem::system(
436        move |engine: Res<LuaEngine>, scripts: Res<ScriptPlugins>, world: &World| {
437            // Dormant fast path: no plugin has resolve hooks (or still needs loading) → no VM hop.
438            let has_work = scripts.plugins.iter().any(|p| {
439                let systems = p.systems.borrow();
440                match &*systems {
441                    LuaPluginSystemsState::NotLoaded => true,
442                    LuaPluginSystemsState::Loaded { .. } => {
443                        !systems.as_loaded().resolve_systems.is_empty()
444                    }
445                    LuaPluginSystemsState::Unloaded => false,
446                }
447            });
448            if !has_work {
449                return;
450            }
451            let timing = world
452                .resources
453                .get::<StageTimings>()
454                .map(|t| t.enabled)
455                .unwrap_or(false);
456            let mut plugin_ns: Vec<u64> =
457                if timing { vec![0; scripts.plugins.len()] } else { Vec::new() };
458            engine.exec(|lua| {
459                Frozen::<Freeze![&'freeze World]>::in_scope(world, |world| {
460                    lua.enter(|ctx| {
461                        let env = ctx.singletons().get(ctx, bindings::env);
462                        let worldref = WorldRef(world);
463                        worldref.add_to_env(ctx, env);
464                    });
465
466                    for (plugin_idx, plugin) in scripts.plugins.iter().enumerate() {
467                        let t0 = timing.then(std::time::Instant::now);
468                        // Lazy-load here too: a resolve batch can arrive before the first
469                        // simulate tick (server boot joins).
470                        if !plugin.has_loaded() {
471                            if let Err(e) = plugin.load(engine.executor().clone(), lua) {
472                                eprintln!("lua plugin load error: {e}");
473                                tracing::error!("Error loading lua plugin: {e}");
474                            }
475                        }
476                        if matches!(&*plugin.systems.borrow(), LuaPluginSystemsState::NotLoaded) {
477                            continue;
478                        }
479                        let systems = plugin.systems.borrow();
480                        let systems = systems.as_loaded();
481                        for closure in &systems.resolve_systems {
482                            let executor = lua.enter(|ctx| {
483                                let closure = ctx.registry().fetch(closure);
484                                let ex =
485                                    crate::lua::piccolo::Executor::start(ctx, closure.into(), ());
486                                ctx.registry().stash(&ctx, ex)
487                            });
488                            if let Err(e) = lua.execute::<()>(&executor) {
489                                eprintln!("lua resolve system error: {e}");
490                                tracing::error!("Error running lua resolve system: {e}");
491                            }
492                        }
493                        if let Some(t0) = t0 {
494                            plugin_ns[plugin_idx] += t0.elapsed().as_nanos() as u64;
495                        }
496                    }
497                })
498            });
499            if timing {
500                if let Some(mut t) = world.resources.get_mut::<StageTimings>() {
501                    for (i, ns) in plugin_ns.iter().enumerate() {
502                        if *ns > 0 {
503                            let label =
504                                scripts.labels.get(i).map(|l| l.as_str()).unwrap_or("?");
505                            t.record("resolve", &format!("lua:{label}"), *ns);
506                        }
507                    }
508                }
509            }
510        },
511    );
512    runner.name = "lua_resolve_hooks";
513    builder.add_system_to_stage(CoreStage::Last, runner);
